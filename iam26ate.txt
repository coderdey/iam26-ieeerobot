# gripper_test.py — run on desktop
from scservo_sdk import PortHandler, PacketHandler
import time

PORT = "/dev/ttyACM1"  # your follower arm port
BAUDRATE = 1_000_000
GRIPPER_ID = 6

ph = PortHandler(PORT)
ph.openPort()
ph.setBaudRate(BAUDRATE)
pk = PacketHandler(0)

# Set slow speed for safety
pk.write2ByteTxRx(ph, GRIPPER_ID, 46, 100)

print("Type a position (1400=open, 5000=closed) or 'q' to quit")
print("Start at 1400 (open), increase slowly until object is gripped firmly")
print("STOP before the servo stalls!\n")

while True:
    val = input("Position > ")
    if val.strip().lower() == 'q':
        break
    try:
        pos = int(val)
        pk.write2ByteTxRx(ph, GRIPPER_ID, 42, pos)
        print(f"  Moved gripper to {pos}")
        time.sleep(0.5)
        # Read current position
        cur_pos, _, _ = pk.read2ByteTxRx(ph, GRIPPER_ID, 56)
        print(f"  Current position: {cur_pos}")
    except ValueError:
        print("  Enter a number or 'q'")

# Open gripper before exiting
pk.write2ByteTxRx(ph, GRIPPER_ID, 42, 1400)
ph.closePort()
print("Done. Gripper opened.")

The joint values are calculated automatically by IK — you don't manually set them. You give move_to() an XYZ position in meters, IK computes the 5 joint angles, and move_joints() sends them to the servos. That's the whole point of IK.

The flow is:
1. Camera sees object at pixel (320, 240)
2. ZED 2 depth gives you 3D position in camera frame
3. T_cam_to_robot transform converts to robot coordinates: e.g. [0.12, 0.08, 0.01] meters
4. solve_ik([0.12, 0.08, 0.01]) → returns 5 joint angles like [33.2, -15.1, 45.3, -60.2, 0.0]
5. move_joints() sends those angles to the servos

The Fix (I'll Write It Now)

The change is straightforward — replace pick_and_place.py's servo calls with LeRobot's interface. The IK math stays the same. Only the "send to servos" part changes:

# unified_pick_place.py
# Uses LeRobot's interface for ALL servo control
# Keeps IK math + detection from pick_and_place.py

from lerobot.robots.so_follower import SO100Follower, SO100FollowerConfig
from lerobot.cameras.opencv import OpenCVCameraConfig
import numpy as np
import time

# Your measured link lengths
BASE_HEIGHT    = 0.045   # UPDATE
SHOULDER_ELBOW = 0.105   # UPDATE
ELBOW_WRIST    = 0.105   # UPDATE
WRIST_GRIPPER  = 0.090   # UPDATE

GRIPPER_OPEN  = 1400
GRIPPER_CLOSE = 5000

GRIP_POSITIONS = {
    "default": 4000,
    # Fill in after today's measurements:
    # ("cube", "small"): 4200,
    # ("cube", "large"): 3600,
}

# Heights in meters above table
APPROACH_HEIGHT = 0.12
GRASP_HEIGHT    = 0.012
LIFT_HEIGHT     = 0.14
DROP_HEIGHT     = 0.05


class UnifiedArm:
    """Wraps LeRobot's SO100Follower but adds IK from pick_and_place.py"""

    def __init__(self, port="/dev/ttyACM1", arm_id="follower_left"):
        # LeRobot connection
        config = SO100FollowerConfig(
            port=port,
            id=arm_id,
            cameras={
                "top": OpenCVCameraConfig(
                    index_or_path=0, width=2560, height=720, fps=30
                ),
            },
        )
        self.robot = SO100Follower(config)
        self.robot.connect()

        # IK chain (same as pick_and_place.py)
        self.chain = self._build_chain()

        # Joint names in LeRobot's format
        self.joint_names = [
            "shoulder_pan", "shoulder_lift", "elbow_flex",
            "wrist_flex", "wrist_roll", "gripper"
        ]

    def _build_chain(self):
        import ikpy.chain as ikchain
        import ikpy.link as iklink
        return ikchain.Chain(name="so100", links=[
            iklink.OriginLink(),
            iklink.URDFLink(name="base",
                translation_vector=[0, 0, BASE_HEIGHT],
                orientation=[0, 0, 0], rotation=[0, 0, 1],
                bounds=np.radians((-135, 135))),
            iklink.URDFLink(name="shoulder",
                translation_vector=[0, 0, 0],
                orientation=[0, 0, 0], rotation=[0, 1, 0],
                bounds=np.radians((-90, 90))),
            iklink.URDFLink(name="elbow",
                translation_vector=[0, 0, SHOULDER_ELBOW],
                orientation=[0, 0, 0], rotation=[0, 1, 0],
                bounds=np.radians((-90, 90))),
            iklink.URDFLink(name="wrist_pitch",
                translation_vector=[0, 0, ELBOW_WRIST],
                orientation=[0, 0, 0], rotation=[0, 1, 0],
                bounds=np.radians((-90, 90))),
            iklink.URDFLink(name="wrist_roll",
                translation_vector=[0, 0, WRIST_GRIPPER],
                orientation=[0, 0, 0], rotation=[0, 0, 1],
                bounds=np.radians((-90, 90))),
        ])

    def solve_ik(self, target_xyz):
        angles_rad = self.chain.inverse_kinematics(np.array(target_xyz))
        angles_deg = np.degrees(angles_rad[1:6])
        angles_deg[3] = -90.0 - angles_deg[1] - angles_deg[2]
        return angles_deg

    def send_joints(self, angles_deg, gripper_pos=None):
        """Send joint angles via LeRobot's interface."""
        action = {}
        for i, name in enumerate(self.joint_names[:5]):
            action[name] = angles_deg[i]
        if gripper_pos is not None:
            action["gripper"] = gripper_pos
        self.robot.send_action(action)

    def move_to(self, xyz, gripper_pos=None):
        angles = self.solve_ik(xyz)
        self.send_joints(angles, gripper_pos)

    def set_gripper(self, state):
        pos = GRIPPER_OPEN if state == "open" else GRIPPER_CLOSE
        action = {name: None for name in self.joint_names[:5]}
        action["gripper"] = pos
        # Only send gripper, read current positions for other joints
        obs = self.robot.get_observation()
        for i, name in enumerate(self.joint_names[:5]):
            action[name] = obs[f"observation.state"][i]
        action["gripper"] = pos
        self.robot.send_action(action)

    def home(self):
        self.send_joints([0, -30, 60, -30, 0], gripper_pos=GRIPPER_OPEN)
        time.sleep(1.0)

    def pick(self, xyz, grip_value=None):
        above = [xyz[0], xyz[1], APPROACH_HEIGHT]
        grasp = [xyz[0], xyz[1], GRASP_HEIGHT]
        grip = grip_value or GRIP_POSITIONS.get("default", 4000)

        self.move_to(above, GRIPPER_OPEN)
        time.sleep(0.8)
        self.move_to(grasp)
        time.sleep(0.6)
        self.send_joints(self.solve_ik(grasp), gripper_pos=grip)
        time.sleep(0.5)
        self.move_to([xyz[0], xyz[1], LIFT_HEIGHT])
        time.sleep(0.6)

    def place(self, bin_xyz):
        above = [bin_xyz[0], bin_xyz[1], LIFT_HEIGHT]
        drop  = [bin_xyz[0], bin_xyz[1], DROP_HEIGHT]

        self.move_to(above)
        time.sleep(0.8)
        self.move_to(drop)
        time.sleep(0.5)
        self.move_to(drop, GRIPPER_OPEN)
        time.sleep(0.3)
        self.move_to(above)
        time.sleep(0.4)

    def get_camera_frame(self):
        obs = self.robot.get_observation()
        return obs.get("observation.images.top")

    def disconnect(self):
        self.robot.disconnect()


# ---- MAIN ----
if __name__ == "__main__":
